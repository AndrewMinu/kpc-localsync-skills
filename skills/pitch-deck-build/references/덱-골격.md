# 피칭덱 골격 (복붙용)

> `pitch-deck-build`가 `pitch-deck.html`을 만들 때 **이 골격 위에** 팀 내용을 얹는다.
> 이동·진행바·편집 로직은 여기 있는 걸 쓰고 **새로 짜지 않는다.** 색·서체는 `design.md`에서 가져온다.

## 왜 이 구조인가

- **화면에 상시로 뜨는 컨트롤 UI가 없다.** 하단 4px 진행바 하나뿐이다. 발표 중에 화살표 버튼이나
  키보드 안내문이 떠 있으면 심사위원 눈이 거기로 간다. 조작법은 발표자만 알면 된다.
- **1920×1080 고정 캔버스를 `scale()`로 축소**한다. 슬라이드가 리플로우되지 않으니 노트북에서 짠 게
  빔프로젝터에서 그대로 나온다. 반응형으로 짜면 화면 비율마다 다르게 깨진다.
  (폰에서 크게 보여야 하는 건 덱이 아니라 `prototype.html` 쪽이다.)
- **인쇄하면 슬라이드 1장 = 1페이지.** `Cmd/Ctrl+P` → PDF로 저장이 그냥 된다. 백업 자료가 공짜로 생긴다.

## HTML 골격

```html
<!-- 고칠 곳: 텍스트는 아래 section.slide 안, 색·크기는 :root -->
<div class="deck-viewport">
  <div class="deck-stage" id="stage">
    <section class="slide"><div class="pageno">1 / 7</div> …표지… </section>
    <section class="slide"><div class="pageno">2 / 7</div> …문제… </section>
    <!-- 슬라이드 1장 = section.slide 1개. 텍스트는 반드시 이 안에 직접 쓴다 -->
  </div>
</div>

<div class="progress" id="progress"></div>
<div class="edit-hotzone" id="editHotzone"></div>
<button class="edit-toggle" id="editToggle" title="편집 (E)">✏️ 편집</button>
```

## JS 골격

```js
const slides=[...document.querySelectorAll('.slide')], stage=document.getElementById('stage');
let cur=0;

// 뷰포트에 맞춰 캔버스 축소 — 이게 있어야 어떤 빔에서도 레이아웃이 같다
const fit=()=>{const f=Math.min(innerWidth/1920,innerHeight/1080);
  stage.style.transform=`translate(${(innerWidth-1920*f)/2}px,${(innerHeight-1080*f)/2}px) scale(${f})`};
fit(); addEventListener('resize',fit);

function show(i){
  cur=Math.max(0,Math.min(i,slides.length-1));
  slides.forEach((s,k)=>{s.classList.toggle('active',k===cur);s.classList.toggle('visible',k===cur)});
  history.replaceState(null,'','#'+(cur+1));
  document.getElementById('progress').style.transform=`scaleX(${(cur+1)/slides.length})`;
}

addEventListener('keydown',e=>{
  if(e.target.isContentEditable)return;           // 편집 중엔 넘기지 않는다
  if(['ArrowRight','ArrowDown',' ','PageDown'].includes(e.key)){e.preventDefault();show(cur+1)}
  else if(['ArrowLeft','ArrowUp','PageUp'].includes(e.key)){e.preventDefault();show(cur-1)}
  else if(e.key==='Home')show(0);
  else if(e.key==='End')show(slides.length-1);
});

let wt=0;                                          // 휠 — 트랙패드가 여러 번 튀는 걸 막는다
addEventListener('wheel',e=>{
  if(document.body.classList.contains('editing'))return;
  const now=Date.now(); if(now-wt<650||Math.abs(e.deltaY)<12)return;
  wt=now; show(cur+(e.deltaY>0?1:-1));
},{passive:true});

let x0=0,y0=0;                                     // 터치 스와이프 — 폰으로 넘겨볼 때
addEventListener('touchstart',e=>{x0=e.touches[0].clientX;y0=e.touches[0].clientY},{passive:true});
addEventListener('touchend',e=>{
  const dx=e.changedTouches[0].clientX-x0, dy=e.changedTouches[0].clientY-y0;
  if(Math.abs(dx)>60&&Math.abs(dx)>Math.abs(dy))show(cur+(dx<0?1:-1));
},{passive:true});

show(Math.max(0,(parseInt(location.hash.slice(1))||1)-1));   // #3 으로 열면 3장부터

/* ── 편집 모드 — 리허설 중 즉석 문구 다듬기용. 저장 기능이 아니다 ── */
const EDIT_SEL='h1,h2,h3,p,li,.pageno~*';
const toggle=document.getElementById('editToggle'), hot=document.getElementById('editHotzone');
let editing=false, hideT=null;
function toggleEdit(){
  editing=!editing;
  document.body.classList.toggle('editing',editing);
  toggle.classList.toggle('active',editing);
  toggle.textContent = editing ? '✏️ 편집 중 · 새로고침하면 사라짐' : '✏️ 편집';
  document.querySelectorAll('.slide '+EDIT_SEL).forEach(el=>{
    editing ? el.setAttribute('contenteditable','true') : el.removeAttribute('contenteditable');
  });
}
toggle.addEventListener('click',toggleEdit);
hot.addEventListener('mouseenter',()=>{clearTimeout(hideT);toggle.classList.add('show')});
[hot,toggle].forEach(el=>el.addEventListener('mouseleave',()=>{
  hideT=setTimeout(()=>{if(!editing)toggle.classList.remove('show')},400)}));
toggle.addEventListener('mouseenter',()=>clearTimeout(hideT));
addEventListener('keydown',e=>{if((e.key==='e'||e.key==='E')&&!e.target.isContentEditable)toggleEdit()});
```

## 넣지 않는 것

- **`localStorage` 자동저장 · 파일 다운로드(Ctrl+S)** — 이 번들은 브라우저 저장소를 금지한다.
  편집은 **리허설 중 즉석 다듬기**용이고, 영구 반영은 `pitch-deck.html`을 직접 고친다.
  팀에게 그렇게 설명한다. 라벨이 `편집 중 · 새로고침하면 사라짐`인 이유다.
- **차트 라이브러리** — 발표장 wifi가 끊기면 데이터 근거 슬라이드만 빈다. 인라인 SVG나 CSS 막대로 그린다.
- **화살표 버튼 · 키보드 안내문 · 페이지 인디케이터 위젯** — 화면에 두지 않는다.
  조작법은 발표자만 알면 되고, 페이지 번호는 슬라이드 안 `.pageno`에 있다.
